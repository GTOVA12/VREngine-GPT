cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(FactoryCore VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
option(FACTORYCORE_ENABLE_LUA "Build Lua control scripting" ON)
option(FACTORYCORE_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(FactoryCoreSimulation
	Source/Simulation/Machine.cpp
	Source/Simulation/Persistence.cpp
	Source/Simulation/Controller.cpp)
add_library(FactoryCore::Simulation ALIAS FactoryCoreSimulation)
target_include_directories(FactoryCoreSimulation PUBLIC Include)
target_compile_features(FactoryCoreSimulation PUBLIC cxx_std_20)
set_target_properties(FactoryCoreSimulation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
	target_compile_options(FactoryCoreSimulation PUBLIC /W4 /WX /permissive- /EHsc /utf-8)
else()
	target_compile_options(FactoryCoreSimulation PUBLIC -Wall -Wextra -Wpedantic -Werror)
	if(FACTORYCORE_ENABLE_SANITIZERS)
		target_compile_options(FactoryCoreSimulation PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
		target_link_options(FactoryCoreSimulation PUBLIC -fsanitize=address,undefined)
	endif()
endif()
add_executable(FactoryCoreRuntime Source/Runtime/Main.cpp)
target_link_libraries(FactoryCoreRuntime PRIVATE FactoryCoreSimulation)
add_executable(FactoryCoreAuthor Source/Author/Main.cpp)
target_link_libraries(FactoryCoreAuthor PRIVATE FactoryCoreSimulation)
if(BUILD_TESTING)
	add_executable(FactoryCoreTests Tests/SimulationTests.cpp)
	target_link_libraries(FactoryCoreTests PRIVATE FactoryCoreSimulation)
	foreach(testName CylinderTravel CylinderPartialStep CylinderFaultLatch ExternalFault EmergencyStop SensorHysteresis ActuatorDelay MotorAcceleration ConveyorWorkpiece ConfigurationValidation SignalValidation ConnectionValidation SignalLatency OrderIndependence FeedbackSampling DisconnectClearsInput RemoveClearsConnections DuplicationResetsState ConfigureResetsState ResetRestoresConfiguration AssemblyRemapping TransactionalConfiguration IdentityValidation PersistenceRoundTrip PersistenceResetsLiveState MalformedConfigurations PersistenceSizeLimit StreamFailures FilePersistence ControllerCycles ControllerFaults DeterministicReplay NumericMotionBound PlacementPreservesState ControllerRejectsWiringAtomically)
		add_test(NAME ${testName} COMMAND FactoryCoreTests ${testName})
	endforeach()
	add_test(NAME RuntimeDemo COMMAND FactoryCoreRuntime --demo)
	add_test(NAME RuntimeFile COMMAND FactoryCoreRuntime "${CMAKE_CURRENT_SOURCE_DIR}/Examples/CylinderCell.factory" 100)
	add_test(NAME AuthoringWorkflow COMMAND ${CMAKE_COMMAND}
		"-DFACTORYCORE_AUTHOR=$<TARGET_FILE:FactoryCoreAuthor>"
		"-DTEST_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}/Testing/Author-$<CONFIG>"
		"-DEXAMPLE=${CMAKE_CURRENT_SOURCE_DIR}/Examples/CylinderCell.factory"
		-P "${CMAKE_CURRENT_SOURCE_DIR}/Tests/AuthoringTests.cmake")
	add_test(NAME AuthorInspect COMMAND FactoryCoreAuthor inspect "${CMAKE_CURRENT_SOURCE_DIR}/Examples/CylinderCell.factory")
	add_test(NAME RuntimeInvalidArguments COMMAND FactoryCoreRuntime --invalid)
	set_tests_properties(RuntimeInvalidArguments PROPERTIES WILL_FAIL TRUE)
endif()
if(FACTORYCORE_ENABLE_LUA)
	include(CMake/Lua.cmake)
endif()
add_library(FactoryCoreEditorModel Source/Editor/EditorSession.cpp Source/Editor/EditorFiles.cpp)
target_link_libraries(FactoryCoreEditorModel PUBLIC FactoryCoreSimulation)
if(BUILD_TESTING)
	add_executable(FactoryCoreEditorTests Tests/EditorSessionTests.cpp)
	target_link_libraries(FactoryCoreEditorTests PRIVATE FactoryCoreEditorModel)
	add_test(NAME EditorSession COMMAND FactoryCoreEditorTests)
endif()
option(FACTORYCORE_ENABLE_EDITOR "Build the Vulkan 3D editor" OFF)
if(FACTORYCORE_ENABLE_EDITOR)
	include(CMake/Editor.cmake)
endif()
install(TARGETS FactoryCoreRuntime FactoryCoreAuthor RUNTIME DESTINATION bin)
install(FILES README.md ThirdPartyNotices.md DESTINATION share/FactoryCore)
install(DIRECTORY Docs/ DESTINATION share/FactoryCore/Docs)
install(DIRECTORY Examples/ DESTINATION share/FactoryCore/Examples)
set(CPACK_PACKAGE_NAME FactoryCore)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
